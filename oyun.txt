<script>
    // Karakter Veritabanı
    const karakterHavuzu = [
        {isim: "Spider-Man", hp_max: 120, hp: 120, saldiri: 25, savunma: 10, super_isim: "Web Blast!", super_bonus: 15, ascii: " /\\_/\\ \n (o.o)\n > ^ <", super_kullanildi: false},
        {isim: "Iron Man", hp_max: 100, hp: 100, saldiri: 30, savunma: 15, super_isim: "Repulsor Beam!", super_bonus: 20, ascii: " [===]\n (o o)\n |===|", super_kullanildi: false},
        {isim: "Wolverine", hp_max: 150, hp: 150, saldiri: 20, savunma: 20, super_isim: "Claw Fury!", super_bonus: 12, ascii: " /_\\_/\n (o.o)\n /| |\\", super_kullanildi: false},
        {isim: "Captain America", hp_max: 130, hp: 130, saldiri: 22, savunma: 12, super_isim: "Shield Bash!", super_bonus: 14, ascii: " [===]\n (o o)\n |===|", super_kullanildi: false},
        {isim: "Thor", hp_max: 140, hp: 140, saldiri: 28, savunma: 8, super_isim: "Lightning Strike!", super_bonus: 30, ascii: " /_\\_/\n (O O)\n |===|", super_kullanildi: false},
        {isim: "Hulk", hp_max: 200, hp: 200, saldiri: 35, savunma: 5, super_isim: "SMASH!", super_bonus: 50, ascii: " /~_~\\\n (O O)\n |===|", super_kullanildi: false},
        {isim: "Batman", hp_max: 140, hp: 140, saldiri: 24, savunma: 20, super_isim: "Batarang Storm!", super_bonus: 25, ascii: " /_\\_/\n (o o)\n |===|", super_kullanildi: false},
        {isim: "Superman", hp_max: 250, hp: 250, saldiri: 40, savunma: 30, super_isim: "Heat Vision!", super_bonus: 45, ascii: " /_\\_/\n (O O)\n |===|", super_kullanildi: false}
    ];

    let gameMode = 1;
    let p1 = null, p2 = null;
    let activePlayer = 1;
    let round = 1;
    let tournamentBracket = [];
    let tournamentStep = 0;

    function showScreen(screenId) {
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        document.getElementById(screenId).classList.add('active');
    }

    function startMode(mode) {
        gameMode = mode;
        p1 = null; p2 = null;
        tournamentBracket = [];
        buildSelectScreen();
        showScreen('select-screen');
    }

    function buildSelectScreen() {
        const grid = document.getElementById('char-grid');
        grid.innerHTML = '';
        const title = document.getElementById('select-title');
        
        if(gameMode === 3) {
            title.innerText = `Turnuva İçin Karakter Seçin (${tournamentBracket.length + 1}/8)`;
        } else {
            title.innerText = !p1 ? "Oyuncu 1 Karakter Seçimi" : "Oyuncu 2 Karakter Seçimi (veya AI)";
        }

        karakterHavuzu.forEach((char, index) => {
            const card = document.createElement('div');
            card.className = 'char-card';
            card.innerHTML = `<strong>${char.isim}</strong><br>HP: ${char.hp_max} | ATK: ${char.saldiri} | DEF: ${char.savunma}`;
            card.onclick = () => selectCharacter(index);
            grid.appendChild(card);
        });
    }

    function selectCharacter(index) {
        let selectedChar = JSON.parse(JSON.stringify(karakterHavuzu[index])); // Deep copy

        if (gameMode === 3) { // Turnuva Modu
            tournamentBracket.push(selectedChar);
            if(tournamentBracket.length < 8) {
                buildSelectScreen();
            } else {
                startTournamentMatch();
            }
        } else { // Tekli veya AI Modu
            if (!p1) {
                p1 = selectedChar;
                if(gameMode === 2) { // AI Modu ise P2'yi rastgele seç
                    let aiIndex = Math.floor(Math.random() * karakterHavuzu.length);
                    p2 = JSON.parse(JSON.stringify(karakterHavuzu[aiIndex]));
                    initBattle();
                } else {
                    buildSelectScreen();
                }
            } else {
                p2 = selectedChar;
                initBattle();
            }
        }
    }

    function startTournamentMatch() {
        p1 = tournamentBracket[tournamentStep];
        p2 = tournamentBracket[tournamentStep + 1];
        logMessage(`--- Turnuva Maçı: ${p1.isim} vs ${p2.isim} ---`);
        initBattle();
    }

    function initBattle() {
        p1.hp = p1.hp_max; p1.super_kullanildi = false;
        p2.hp = p2.hp_max; p2.super_kullanildi = false;
        round = 1;
        activePlayer = 1;
        
        document.getElementById('p1-name').innerText = p1.isim;
        document.getElementById('p2-name').innerText = p2.isim;
        document.getElementById('p1-ascii').innerText = p1.ascii;
        document.getElementById('p2-ascii').innerText = p2.ascii;
        
        document.getElementById('battle-log').innerHTML = 'Savaş başladı!<br>';
        updateUI();
        showScreen('battle-screen');
    }

    function updateUI() {
        document.getElementById('round-title').innerText = `TUR ${round}`;
        
        // Can barları kontrolü
        const p1Pct = Math.max(0, (p1.hp / p1.hp_max) * 100);
        const p2Pct = Math.max(0, (p2.hp / p2.hp_max) * 100);
        document.getElementById('p1-bar').style.width = p1Pct + '%';
        document.getElementById('p2-bar').style.width = p2Pct + '%';
        
        document.getElementById('p1-hp-text').innerText = `${p1.hp}/${p1.hp_max} HP`;
        document.getElementById('p2-hp-text').innerText = `${p2.hp}/${p2.hp_max} HP`;

        // Butonların aktiflik durumları (Her 3 turda bir Heal)
        document.getElementById('btn-heal').disabled = (round % 3 !== 0);
        
        if (activePlayer === 1) {
            document.getElementById('turn-indicator').innerText = `Sıra: ${p1.isim} (P1)`;
            document.getElementById('btn-super').disabled = p1.super_kullanildi;
        } else {
            document.getElementById('turn-indicator').innerText = `Sıra: ${p2.isim} (P2)`;
            document.getElementById('btn-super').disabled = p2.super_kullanildi;
        }
    }

    function logMessage(msg) {
        const logBox = document.getElementById('battle-log');
        logBox.innerHTML += msg + '<br>';
        logBox.scrollTop = logBox.scrollHeight;
    }

    function playTurn(action) {
        let attacker = activePlayer === 1 ? p1 : p2;
        let defender = activePlayer === 1 ? p2 : p1;

        if (action === 'normal') {
            let dmg = Math.max(5, attacker.saldiri - defender.savunma);
            defender.hp -= dmg;
            logMessage(`💥 ${attacker.isim} normal vuruş yaptı! ${defender.isim} ${dmg} hasar aldı.`);
        } 
        else if (action === 'super') {
            let dmg = Math.max(5, (attacker.saldiri + attacker.super_bonus) - defender.savunma);
            defender.hp -= dmg;
            attacker.super_kullanildi = true;
            logMessage(`🔥 ${attacker.isim} ÖZEL YETENEK KULLANDI: [${attacker.super_isim}]! ${defender.isim} ${dmg} hasar aldı.`);
        } 
        else if (action === 'heal') {
            let heal = Math.floor(attacker.hp_max * 0.3);
            attacker.hp = Math.min(attacker.hp_max, attacker.hp + heal);
            logMessage(`💚 ${attacker.isim} iyileşme iksiri içti! +${heal} HP.`);
        }

        if (defender.hp <= 0) {
            defender.hp = 0;
            updateUI();
            endBattle(attacker);
            return;
        }

        // Sıra Değiştirme
        if (gameMode === 2 && activePlayer === 1) {
            // Yapay zeka modu ise oyuncudan hemen sonra AI oynasın
            activePlayer = 2;
            updateUI();
            setTimeout(aiTurn, 1000); // 1 saniye bekleme efekti
        } else {
            activePlayer = activePlayer === 1 ? 2 : 1;
            if (activePlayer === 1) round++;
            updateUI();
        }
    }

    // Kodundaki Akıllı Yapay Zeka Mantığı
    function aiTurn() {
        let superHasar = Math.max(5, (p2.saldiri + p2.super_bonus) - p1.savunma);
        
        if (round % 3 === 0 && p2.hp < p2.hp_max * 0.4) {
            playTurn('heal');
        } else if (!p2.super_kullanildi && p1.hp <= superHasar) {
            playTurn('super');
        } else if (!p2.super_kullanildi && Math.random() < 0.3) {
            playTurn('super');
        } else {
            playTurn('normal');
        }

        if (p1.hp > 0) {
            activePlayer = 1;
            round++;
            updateUI();
        }
    }

    function endBattle(winner) {
        alert(`🏆 Maçın Kazananı: ${winner.isim}!`);
        
        if (gameMode === 3) { // Turnuva Modu İlerlemesi
            tournamentBracket[tournamentStep] = winner; // Kazanan üst tura çıkar
            tournamentBracket.splice(tournamentStep + 1, 1); // Eleneni diziden sil
            
            if (tournamentBracket.length === 1) {
                logMessage(`🏆🏆🏆 TURNUVA ŞAMPİYONU: ${tournamentBracket[0].isim} 🏆🏆🏆`);
                alert(`🏆 TURNUVA ŞAMPİYONU: ${tournamentBracket[0].isim} 🏆`);
                showScreen('menu-screen');
            } else {
                tournamentStep++;
                if (tournamentStep >= tournamentBracket.length) {
                    tournamentStep = 0; // Tur bitti, bir üst tur maçlarına başla
                }
                startTournamentMatch();
            }
        } else {
            showScreen('menu-screen');
        }
    }
</script>
